option('nvcc_threads',
    type: 'integer',
    min: 1,
    max: 32,
    value: 4,
    description: 'nvcc --threads value for the per-kernel fatbin compiles. nvcc parallelises across the gencode architectures (6 of them), so this shortens the critical-path kernel: measured 8.5s -> 2.1s on adm_cm.cu at 4, with all 22 fatbins byte-identical to a serial build. Note ninja already builds the 22 fatbins concurrently, so the effective thread count is this value times the ninja job count -- lower it on a small runner. 1 disables (nvcc ignores --threads 1). See ADR-1224.')

option('compress_device_code',
    type: 'boolean',
    value: true,
    description: 'Store the GPU device code of every backend compressed at the toolchain\'s strongest setting (ADR-1590): nvcc --compress-mode=size with every fatbin entry compressed, hipcc and icpx --offload-compress at zstd level 22. The device code itself is unchanged; the driver or runtime decompresses it when it loads a module. Configure fails when a device compiler in use cannot compress (the clang CUDA driver of enable_nvcc=false, AdaptiveCpp): set false there, or to keep raw device images for inspection.')

option('hip_gfx_targets',
    type: 'string',
    value: '',
    description: 'Comma-separated list of AMD GFX targets for hipcc --offload-arch (e.g. "gfx1036,gfx1100"). When empty (default), auto-detected from rocm_agent_enumerator, then hipconfig; falls back to "gfx90a,gfx1030,gfx1036,gfx1100" (CDNA2 + RDNA2 desktop + Raphael APU iGPU + RDNA3) if both fail — the typical no-GPU build sandbox case.
